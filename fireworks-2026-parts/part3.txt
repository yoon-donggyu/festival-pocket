const KMA_WEATHER_PROXY_URL='https://script.google.com/macros/s/AKfycbxjq6tyDI5iHyXHChKhd4wvsNf42QvBqFI6DNl6oYc1CAEbVFaD5NOSvO2cZoXX856oiw/exec';
let weatherJsonpSeq=0;
const SEOUL_FIREWORKS={lat:37.5285,lng:126.9330,start:'2026-09-05'};
function jsonpKmaWeather(){
return new Promise((resolve,reject)=>{
const cb='fireworksKma_'+Date.now()+'_'+(++weatherJsonpSeq);
const sc=document.createElement('script');
let finished=false;
const cleanup=()=>{
if(finished)return; finished=true;
try{delete window[cb]}catch(e){}
if(sc.parentNode)sc.parentNode.removeChild(sc);
};
window[cb]=(data)=>{cleanup();resolve(data)};
sc.onerror=()=>{cleanup();reject(new Error('기상청 중계 연결 실패'))};
setTimeout(()=>{if(!finished){cleanup();reject(new Error('기상청 응답 시간 초과'))}},12000);
sc.src=KMA_WEATHER_PROXY_URL+'?action=weather&lat='+SEOUL_FIREWORKS.lat+'&lon='+SEOUL_FIREWORKS.lng+'&callback='+encodeURIComponent(cb)+'&_='+Date.now();
document.head.appendChild(sc);
});
}
function fmtHour(t){
const m=String(t||'').match(/(\d{2}):?(\d{2})/);
return m?Number(m[1])+'시':String(t||'');
}
function weatherAdviceText(data){
const c=data.current||{}, ultra=Array.isArray(data.ultraForecast)?data.ultraForecast:[];
const rainy=ultra.some(x=>Number(x.pop)>=40 || /비|눈|소나기/.test(x.sky||'')) || Number(c.precipitation)>0;
const windy=Number(c.windSpeed)>=6;
const temp=Number(c.temp);
let arr=[];
if(rainy)arr.push('우비 4개 필수');
else arr.push('현재 기준 우비는 비상용');
if(windy)arr.push('돗자리 고정 강화');
if(Number.isFinite(temp)&&temp>=27)arr.push('낮 더위 대비 물·선크림');
arr.push('밤에는 얇은 겉옷');
return arr.join(' · ');
}
async function loadLiveWeather(force=false){
const btn=document.getElementById('weatherRefreshBtn');
if(btn){btn.disabled=true;btn.textContent='갱신 중...'}
try{
const data=await jsonpKmaWeather();
const c=data.current||{};
document.getElementById('wTemp').textContent=c.temp!=null?Math.round(Number(c.temp))+'°':'--°';
document.getElementById('wCond').textContent=c.condition||'실황';
document.getElementById('wWind').textContent=c.windSpeed!=null?Number(c.windSpeed).toFixed(1)+' m/s':'--';
document.getElementById('wUpdated').textContent='기상청 APIHub · '+(c.observedAt?String(c.observedAt).replace('T',' '):(data.queriedAt||'최신 조회'))+' · 여의도 좌표 기준';
const ultra=Array.isArray(data.ultraForecast)?data.ultraForecast.slice(0,8):[];
document.getElementById('hourlyWeather').innerHTML=ultra.length?
'<div style="display:flex;gap:8px;overflow-x:auto;padding-bottom:4px">'+ultra.map(r=>
'<div class="metric" style="min-width:105px"><b style="font-size:16px">'+fmtHour(r.time)+'</b><small>'+(r.sky||'-')+'<br>'+(r.temp!=null?Math.round(Number(r.temp))+'°':'')+(r.pop!=null?' · 강수 '+r.pop+'%':'')+'</small></div>'
).join('')+'</div>'
:'초단기예보 데이터가 없습니다.';
const daily=Array.isArray(data.dailyForecast)?data.dailyForecast:[];
const d=daily.find(x=>String(x.date||'').startsWith('2026-09-05'));
if(d){
const min=d.minTemp??d.tmn, max=d.maxTemp??d.tmx, pop=d.maxPop??d.pop;
document.getElementById('eventDayWeather').innerHTML=
'<b>9/5 토요일</b><br>'+(d.condition||d.sky||'예보')+
(min!=null&&max!=null?' · '+Math.round(Number(min))+'~'+Math.round(Number(max))+'℃':'')+
(pop!=null?' · 최대 강수확률 '+Math.round(Number(pop))+'%':'');
}else{
document.getElementById('eventDayWeather').innerHTML='행사일 상세예보가 아직 응답에 포함되지 않았습니다. 날짜가 가까워지면 자동 표시됩니다.';
}
document.getElementById('weatherAdvice').textContent=weatherAdviceText(data);
}catch(err){
document.getElementById('wCond').textContent='조회 실패';
document.getElementById('wUpdated').textContent='기상청 연동 실패 · '+String(err.message||err);
document.getElementById('hourlyWeather').textContent='날씨를 불러오지 못했습니다. 잠시 뒤 새로고침하세요.';
}finally{
if(btn){btn.disabled=false;btn.textContent='기상청 최신값 새로고침'}
}
}
const cbs=[...document.querySelectorAll('#checklist input')];

document.getElementById('prog').style.width=pct+'%';
document.getElementById('progText').textContent=pct+'% 준비 완료 · '+done+'/'+cbs.length;
}

cb.checked=localStorage.getItem('fireworks_'+cb.dataset.key)==='1';
cb.addEventListener('change',()=>{localStorage.setItem('fireworks_'+cb.dataset.key,cb.checked?'1':'0');updateProg()})
});
updateProg();
setTimeout(()=>loadLiveWeather().catch(()=>{}),500);
