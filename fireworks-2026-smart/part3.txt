function seoulParts(d=new Date()){
  const p=new Intl.DateTimeFormat('en-CA',{timeZone:'Asia/Seoul',year:'numeric',month:'2-digit',day:'2-digit',hour:'2-digit',minute:'2-digit',hour12:false}).formatToParts(d);
  return Object.fromEntries(p.map(x=>[x.type,x.value]));
}
function seoulMinutes(d=new Date()){const p=seoulParts(d);return (+p.hour)*60+(+p.minute)}
function seoulDate(d=new Date()){const p=seoulParts(d);return `${p.year}-${p.month}-${p.day}`}
function fmtDuration(ms){
  if(ms<=0)return '진행 중';
  const d=Math.floor(ms/86400000),h=Math.floor(ms%86400000/3600000),m=Math.floor(ms%3600000/60000);
  return d>0?`D-${d}`:`${h}시간 ${m}분`;
}

  if(m<1020)return ['자리로 돌아가세요','이제부터 이동량을 줄이는 게 편합니다.','혼잡'];
  if(m<1110)return ['식사·화장실 먼저','18시 이후 줄이 길어지기 전에 한 번 다녀오기.','혼잡'];

function crowdByTime(){
  if(seoulDate()!==EVENT_DATE)return '낮음';
  const m=seoulMinutes();
  if(m<720)return '낮음'; if(m<960)return '보통'; if(m<1140)return '높음'; return '매우 높음';
}
function updateClock(){
  const now=new Date(), p=seoulParts(now);
  document.getElementById('liveClock').textContent=`${p.hour}:${p.minute}`;
  const diff=START-now;
  const cd=document.getElementById('countdown');
  if(now<START)cd.textContent=fmtDuration(diff);
  else if(now<=END)cd.textContent='진행 중';
  else cd.textContent='종료';
  const [t,d,s]=currentAction();
  document.getElementById('nowTitle').textContent=t;
  document.getElementById('nowDesc').textContent=d;
  document.getElementById('crowdStatus').textContent=s;
  document.getElementById('fieldCrowd').textContent=crowdByTime();
  const dayStart=390,dayEnd=1300,cur=seoulMinutes();
  document.getElementById('dayProgress').style.width=(seoulDate()===EVENT_DATE?Math.max(0,Math.min(100,(cur-dayStart)/(dayEnd-dayStart)*100)):0)+'%';
  const f=document.getElementById('fieldCountdown');
  if(now<START)f.textContent=fmtDuration(diff);
  else if(now<=END){f.textContent='LIVE';document.getElementById('fieldCaption').textContent='지금 불꽃 진행 중'}
  else{f.textContent='END';document.getElementById('fieldCaption').textContent='짐 정리 후 분산 귀가'}
  updateTimeline();
}
function updateTimeline(){
  const nowM=seoulMinutes(), eventToday=seoulDate()===EVENT_DATE;
  const rows=[...document.querySelectorAll('.trow')];
  rows.forEach(r=>r.classList.remove('done','current'));
  if(!eventToday)return;
  let last=null;
  rows.forEach(r=>{
    const [h,m]=r.dataset.time.split(':').map(Number),rm=h*60+m;
    if(nowM>=rm){r.classList.add('done');last=r}
  });
  if(last){last.classList.remove('done');last.classList.add('current')}
}

function jsonpWeather(){
  return new Promise((resolve,reject)=>{
    const cb='kma_'+Date.now()+'_'+Math.random().toString(36).slice(2);
    const s=document.createElement('script');
    const timer=setTimeout(()=>{cleanup();reject(new Error('timeout'))},10000);
    function cleanup(){clearTimeout(timer);delete window[cb];s.remove()}
    window[cb]=data=>{cleanup();resolve(data)};
    s.onerror=()=>{cleanup();reject(new Error('network'))};
    s.src=`${KMA_WEATHER_PROXY_URL}?action=weather&lat=${POS.yeouido[0]}&lon=${POS.yeouido[1]}&callback=${cb}&_=${Date.now()}`;
    document.head.appendChild(s);
  });
}
function extractWeather(data){
  const flat=JSON.stringify(data);
  const temp=(data&& (data.temp??data.temperature??data.current?.temp??data.current?.temperature));
  const wind=(data&& (data.windSpeed??data.wind??data.current?.windSpeed));
  let desc=data?.condition??data?.sky??data?.current?.condition??'기상청';
  return {temp,wind,desc};
}
jsonpWeather().then(data=>{
  const w=extractWeather(data);
  if(w.temp!==undefined&&w.temp!==null)document.getElementById('weatherTemp').textContent=`${Math.round(Number(w.temp))}°`;
  document.getElementById('weatherText').textContent=String(w.desc||'기상청');
  document.getElementById('windValue').textContent=w.wind!==undefined?`${w.wind} m/s`:'확인';
}).catch(()=>{
  document.getElementById('weatherText').textContent='새로고침 시 재시도';
  document.getElementById('windValue').textContent='미수신';
});

function savedSpot(){try{return JSON.parse(localStorage.getItem('fireworks_spot_v2')||'null')}catch{return null}}
document.getElementById('saveSpot').addEventListener('click',()=>{
  const st=document.getElementById('spotState');
  if(!navigator.geolocation){st.textContent='이 브라우저는 위치 저장을 지원하지 않습니다.';return}
  st.textContent='현재 위치 확인 중…';
  navigator.geolocation.getCurrentPosition(p=>{
    const v={lat:p.coords.latitude,lng:p.coords.longitude,time:new Date().toISOString()};
    localStorage.setItem('fireworks_spot_v2',JSON.stringify(v));
    st.textContent=`저장 완료 · ${v.lat.toFixed(5)}, ${v.lng.toFixed(5)}`;
  },()=>st.textContent='위치 권한이 꺼져 있습니다. 브라우저 사이트 권한에서 위치를 허용하세요.',{enableHighAccuracy:true,timeout:12000});
});
document.getElementById('openSpot').addEventListener('click',()=>{
  const v=savedSpot(),st=document.getElementById('spotState');
  if(!v){st.textContent='저장된 위치가 없습니다.';return}
  window.open(`https://map.kakao.com/link/map/우리자리,${v.lat},${v.lng}`,'_blank');
});

});
